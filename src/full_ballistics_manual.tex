\section{Introduction}
This manual presents a complete modern framework for exterior ballistics, including:
\begin{itemize}
\item Point-mass and 6-DOF models
\item Doppler-based drag modeling
\item Numerical integration methods
\item Practical shooting outputs (DOPE)
\end{itemize}

\section{Equations of Motion}
\[
\dot{\bm{r}} = \bm{v}
\]
\[
\dot{\bm{v}} = \frac{1}{m}(\bm{F}_{drag} + \bm{F}_{gravity} + \bm{F}_{coriolis})
\]

\section{Drag Model}
\[
\bm{F}_D = -\frac{1}{2} \rho v^2 C_D(M)\hat{v}
\]

\section{6-DOF Model}
\[
\dot{\bm{\omega}} = I^{-1}(M - \omega \times I\omega)
\]

\section{Doppler Data}
Drag coefficients are obtained from radar data and interpolated via splines.

\section{DOPE}
\[
MIL = \frac{\Delta y}{x} \times 1000
\]

\section{Conclusion}
This document provides a full pipeline from physics to practical shooting solutions.
